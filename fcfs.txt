#include <stdio.h>

int main() {
    int ref[50], frame[10];
    int n, f, i, j, k = 0, faults = 0, found;

    printf("Enter number of pages: ");
    scanf("%d", &n);
    printf("Enter reference string: ");
    for (i = 0; i < n; i++)
        scanf("%d", &ref[i]);
    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &f);
    while (f < 3) {
        printf("Frames must be at least 3. Enter again: ");
        scanf("%d", &f);
    }

    for (i = 0; i < f; i++)
        frame[i] = -1;

    printf("\n--- FCFS ---\n");
    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < f; j++) {
            if (frame[j] == ref[i])
                found = 1;
        }

        if (found == 0) {
            frame[k] = ref[i];
            k = (k + 1) % f;
            faults++;
        }

        printf("%d : ", ref[i]);
        for (j = 0; j < f; j++) {
            if (frame[j] == -1)
                printf("- ");
            else
                printf("%d ", frame[j]);
        }
        if (found == 0)
            printf("  Fault\n");
        else
            printf("  Hit\n");
    }
    printf("Total Page Faults = %d\n", faults);
    return 0;
}
