#include <stdio.h>
#include <stdlib.h>

int req[50], n, head, size;

void sstf() {
    int done[50] = {0};
    int i, j, cur = head, total = 0, min, pos;

    printf("\nSSTF Path: %d", cur);
    for (i = 0; i < n; i++) {
        min = 9999;
        for (j = 0; j < n; j++) {
            if (done[j] == 0 && abs(req[j] - cur) < min) {
                min = abs(req[j] - cur);
                pos = j;
            }
        }
        done[pos] = 1;
        total = total + min;
        cur = req[pos];
        printf(" -> %d", cur);
    }
    printf("\nTotal Head Movement (SSTF) = %d\n", total);
}

void scan() {
    int s[50], i, j, temp, cur = head, total = 0;

    for (i = 0; i < n; i++)
        s[i] = req[i];

    // sort karo (chhota se bada)
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (s[j] > s[j + 1]) {
                temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }

    // head se bada pehla request dhundo
    i = 0;
    while (i < n && s[i] < head)
        i++;

    printf("\nSCAN Path: %d", cur);

    // upar ki taraf (spindle se door)
    for (j = i; j < n; j++) {
        total = total + abs(s[j] - cur);
        cur = s[j];
        printf(" -> %d", cur);
    }

    // disk ke end tak jao
    total = total + abs(size - 1 - cur);
    cur = size - 1;
    printf(" -> %d", cur);

    // wapas neeche ki taraf
    for (j = i - 1; j >= 0; j--) {
        total = total + abs(s[j] - cur);
        cur = s[j];
        printf(" -> %d", cur);
    }
    printf("\nTotal Head Movement (SCAN) = %d\n", total);
}

void clook() {
    int s[50], i, j, temp, cur = head, total = 0;

    for (i = 0; i < n; i++)
        s[i] = req[i];

    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (s[j] > s[j + 1]) {
                temp = s[j];
                s[j] = s[j + 1];
                s[j + 1] = temp;
            }

    i = 0;
    while (i < n && s[i] < head)
        i++;

    printf("\nC-LOOK Path: %d", cur);

    // upar ki taraf, sabse bade request tak
    for (j = i; j < n; j++) {
        total = total + abs(s[j] - cur);
        cur = s[j];
        printf(" -> %d", cur);
    }

    // sabse chhote request pe jump, phir upar chalo
    for (j = 0; j < i; j++) {
        total = total + abs(s[j] - cur);
        cur = s[j];
        printf(" -> %d", cur);
    }
    printf("\nTotal Head Movement (C-LOOK) = %d\n", total);
}

int main() {
    int i;

    printf("Enter number of requests: ");
    scanf("%d", &n);
    printf("Enter request queue: ");
    for (i = 0; i < n; i++)
        scanf("%d", &req[i]);
    printf("Enter initial head position: ");
    scanf("%d", &head);
    printf("Enter disk size (total tracks): ");
    scanf("%d", &size);

    sstf();
    scan();
    clook();

    return 0;
}
