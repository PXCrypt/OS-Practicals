#include <stdio.h>
#include <stdlib.h>

int main() {
    int req[50], n, head, size, i, j, temp, total = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);
    printf("Enter request queue: ");
    for (i = 0; i < n; i++)
        scanf("%d", &req[i]);
    printf("Enter initial head position: ");
    scanf("%d", &head);
    printf("Enter disk size (total tracks): ");
    scanf("%d", &size);

    // sort karo (chhota se bada)
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - i - 1; j++)
            if (req[j] > req[j + 1]) {
                temp = req[j];
                req[j] = req[j + 1];
                req[j + 1] = temp;
            }

    // head se bada pehla request dhundo
    i = 0;
    while (i < n && req[i] < head)
        i++;

    printf("\nSCAN Path: %d", head);

    // upar ki taraf (spindle se door)
    for (j = i; j < n; j++) {
        total = total + abs(req[j] - head);
        head = req[j];
        printf(" -> %d", head);
    }

    // disk ke end tak jao
    total = total + abs(size - 1 - head);
    head = size - 1;
    printf(" -> %d", head);

    // wapas neeche ki taraf
    for (j = i - 1; j >= 0; j--) {
        total = total + abs(req[j] - head);
        head = req[j];
        printf(" -> %d", head);
    }
    printf("\nTotal Head Movement = %d\n", total);
    return 0;
}
