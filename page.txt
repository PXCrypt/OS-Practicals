#include <stdio.h>

int n, f;        // n = pages ki sankhya, f = frames ki sankhya
int ref[50];     // reference string

// Frames ko print karne ke liye
void show(int frame[]) {
    int j;
    for (j = 0; j < f; j++) {
        if (frame[j] == -1)
            printf("- ");
        else
            printf("%d ", frame[j]);
    }
}

// ---------- FCFS ----------
void fcfs() {
    int frame[10], i, j, k = 0, faults = 0, found;

    for (i = 0; i < f; i++)
        frame[i] = -1;

    printf("\n----- FCFS -----\n");
    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < f; j++) {
            if (frame[j] == ref[i])
                found = 1;
        }

        if (found == 0) {
            frame[k] = ref[i];
            k = (k + 1) % f;
            faults++;
        }

        printf("%d : ", ref[i]);
        show(frame);
        if (found == 0)
            printf("  Fault\n");
        else
            printf("  Hit\n");
    }
    printf("Total Page Faults (FCFS) = %d\n", faults);
}

// ---------- LRU ----------
void lru() {
    int frame[10], used[10], i, j, pos, faults = 0, found;

    for (i = 0; i < f; i++) {
        frame[i] = -1;
        used[i] = 0;
    }

    printf("\n----- LRU -----\n");
    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < f; j++) {
            if (frame[j] == ref[i]) {
                found = 1;
                used[j] = i + 1;      // abhi use hua, time update
            }
        }

        if (found == 0) {
            pos = -1;
            // pehle khaali frame dhundo
            for (j = 0; j < f; j++) {
                if (frame[j] == -1) {
                    pos = j;
                    break;
                }
            }
            // khaali nahi mila to sabse purana use wala dhundo
            if (pos == -1) {
                pos = 0;
                for (j = 1; j < f; j++) {
                    if (used[j] < used[pos])
                        pos = j;
                }
            }
            frame[pos] = ref[i];
            used[pos] = i + 1;
            faults++;
        }

        printf("%d : ", ref[i]);
        show(frame);
        if (found == 0)
            printf("  Fault\n");
        else
            printf("  Hit\n");
    }
    printf("Total Page Faults (LRU) = %d\n", faults);
}

// ---------- OPTIMAL ----------
void optimal() {
    int frame[10], i, j, k, pos, faults = 0, found, next, max;

    for (i = 0; i < f; i++)
        frame[i] = -1;

    printf("\n----- OPTIMAL -----\n");
    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < f; j++) {
            if (frame[j] == ref[i])
                found = 1;
        }

        if (found == 0) {
            pos = -1;
            // pehle khaali frame dhundo
            for (j = 0; j < f; j++) {
                if (frame[j] == -1) {
                    pos = j;
                    break;
                }
            }
            // khaali nahi mila to future dekho
            if (pos == -1) {
                max = -1;
                for (j = 0; j < f; j++) {
                    next = n;    // maan lo ye page future mein aayega hi nahi
                    for (k = i + 1; k < n; k++) {
                        if (ref[k] == frame[j]) {
                            next = k;
                            break;
                        }
                    }
                    if (next > max) {
                        max = next;
                        pos = j;
                    }
                }
            }
            frame[pos] = ref[i];
            faults++;
        }

        printf("%d : ", ref[i]);
        show(frame);
        if (found == 0)
            printf("  Fault\n");
        else
            printf("  Hit\n");
    }
    printf("Total Page Faults (Optimal) = %d\n", faults);
}

// ---------- MAIN ----------
int main() {
    int i;

    printf("Enter number of pages: ");
    scanf("%d", &n);
    printf("Enter reference string: ");
    for (i = 0; i < n; i++)
        scanf("%d", &ref[i]);

    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &f);
    while (f < 3) {
        printf("Frames must be at least 3. Enter again: ");
        scanf("%d", &f);
    }

    fcfs();
    lru();
    optimal();

    return 0;
}
