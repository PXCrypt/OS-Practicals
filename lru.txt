#include <stdio.h>

int main() {
    int ref[50], frame[10], time[10];
    int n, f, i, j, pos, faults = 0, found;

    printf("Enter number of pages: ");
    scanf("%d", &n);
    printf("Enter reference string: ");
    for (i = 0; i < n; i++)
        scanf("%d", &ref[i]);
    printf("Enter number of frames (minimum 3): ");
    scanf("%d", &f);
    while (f < 3) {
        printf("Frames must be at least 3. Enter again: ");
        scanf("%d", &f);
    }

    for (i = 0; i < f; i++) {
        frame[i] = -1;
        time[i] = 0;
    }

    printf("\n--- LRU ---\n");
    for (i = 0; i < n; i++) {
        found = 0;
        for (j = 0; j < f; j++) {
            if (frame[j] == ref[i]) {
                found = 1;
                time[j] = i + 1;      // abhi use hua
            }
        }

        if (found == 0) {
            pos = -1;
            for (j = 0; j < f; j++) {     // khaali frame dhundo
                if (frame[j] == -1) {
                    pos = j;
                    break;
                }
            }
            if (pos == -1) {              // khaali nahi hai
                pos = 0;
                for (j = 1; j < f; j++) {
                    if (time[j] < time[pos])
                        pos = j;
                }
            }
            frame[pos] = ref[i];
            time[pos] = i + 1;
            faults++;
        }

        printf("%d : ", ref[i]);
        for (j = 0; j < f; j++) {
            if (frame[j] == -1)
                printf("- ");
            else
                printf("%d ", frame[j]);
        }
        if (found == 0)
            printf("  Fault\n");
        else
            printf("  Hit\n");
    }
    printf("Total Page Faults = %d\n", faults);
    return 0;
}
