#include <stdio.h>
#include <stdlib.h>

int main() {
    int req[50], done[50] = {0};
    int n, head, i, j, pos, min, total = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);
    printf("Enter request queue: ");
    for (i = 0; i < n; i++)
        scanf("%d", &req[i]);
    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("\nSSTF Path: %d", head);
    for (i = 0; i < n; i++) {
        min = 9999;
        for (j = 0; j < n; j++) {
            if (done[j] == 0 && abs(req[j] - head) < min) {
                min = abs(req[j] - head);
                pos = j;
            }
        }
        done[pos] = 1;
        total = total + min;
        head = req[pos];
        printf(" -> %d", head);
    }
    printf("\nTotal Head Movement = %d\n", total);
    return 0;
}
