#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

int data = 0;          // shared data
int readcount = 0;

pthread_mutex_t mutex;   // readcount ke liye
sem_t wrt;               // writer ke liye

void *reader(void *arg) {
    int id = *(int *)arg;

    pthread_mutex_lock(&mutex);
    readcount++;
    if (readcount == 1)
        sem_wait(&wrt);          // pehla reader writer ko rokta hai
    pthread_mutex_unlock(&mutex);

    printf("Reader %d is reading data = %d\n", id, data);
    sleep(1);

    pthread_mutex_lock(&mutex);
    readcount--;
    if (readcount == 0)
        sem_post(&wrt);          // aakhri reader writer ko chhodta hai
    pthread_mutex_unlock(&mutex);

    return NULL;
}

void *writer(void *arg) {
    int id = *(int *)arg;

    sem_wait(&wrt);
    data = data + 10;
    printf("Writer %d wrote data = %d\n", id, data);
    sleep(1);
    sem_post(&wrt);

    return NULL;
}

int main() {
    pthread_t r[3], w[2];
    int rid[3] = {1, 2, 3};
    int wid[2] = {1, 2};
    int i;

    pthread_mutex_init(&mutex, NULL);
    sem_init(&wrt, 0, 1);

    pthread_create(&r[0], NULL, reader, &rid[0]);
    pthread_create(&w[0], NULL, writer, &wid[0]);
    pthread_create(&r[1], NULL, reader, &rid[1]);
    pthread_create(&r[2], NULL, reader, &rid[2]);
    pthread_create(&w[1], NULL, writer, &wid[1]);

    for (i = 0; i < 3; i++)
        pthread_join(r[i], NULL);
    for (i = 0; i < 2; i++)
        pthread_join(w[i], NULL);

    sem_destroy(&wrt);
    pthread_mutex_destroy(&mutex);

    printf("\nAll done.\n");
    return 0;
}
